import { useState, useEffect } from "react";
import { Text, Box, render } from "ink";
import { memoryUsage } from "node:process";
import type { Stats } from "./Stats";

export function formatBytes(n: number): string {
  if (n < 1024) return `${n.toFixed(0)} B`;
  if (n < 1024 ** 2) return `${(n / 1024).toFixed(1)} KB`;
  if (n < 1024 ** 3) return `${(n / 1024 ** 2).toFixed(2)} MB`;
  return `${(n / 1024 ** 3).toFixed(2)} GB`;
}

function formatMbps(bytesPerSecond: number): string {
  return `${((bytesPerSecond * 8) / 1e6).toFixed(2)} Mbps`;
}

function formatEta(sec: number): string {
  if (!isFinite(sec)) return "--";
  if (sec === 0) return "done 🎉";
  const h = Math.floor(sec / 3600);
  const m = Math.floor((sec % 3600) / 60);
  const s = Math.floor(sec % 60);
  return h > 0 ? `${h}h ${m}m` : m > 0 ? `${m}m ${s}s` : `${s}s`;
}

function renderProgressBar(progress: number, width: number = 52): string {
  const filledLength = Math.round(width * Math.min(Math.max(progress, 0), 1));
  const emptyLength = width - filledLength;
  return "█".repeat(filledLength) + "░".repeat(emptyLength);
}

function App({ stats }: { stats: Stats }) {
  const [, forceTick] = useState(0);
  const delayMs = 500;

  useEffect(() => {
    const id = setInterval(() => {
      stats.sample(delayMs);
      forceTick((t) => t + 1);
    }, delayMs);
    return () => clearInterval(id);
  }, [stats]);

  const pct = (stats.progress * 100).toFixed(1);
  const memory = memoryUsage();
  const topPeers = stats.topPeers(6);

  return (
    <Box flexDirection="column" paddingY={1} paddingX={2} width={70}>
      {/* ── PROGRESS BAR HEADER ── */}
      <Box marginBottom={1} flexDirection="column">
        <Box justifyContent="space-between" width={52}>
          <Text bold color="greenBright">
            Downloading Torrent
          </Text>
          <Text bold color="greenBright">
            {pct}%
          </Text>
        </Box>
        <Text color="green">{renderProgressBar(stats.progress, 52)}</Text>
      </Box>

      {/* ── METRICS GRID ── */}
      <Box flexDirection="column" marginBottom={1}>
        <Box>
          <Text>Speed </Text>
          <Text color="greenBright" bold>
            {formatMbps(stats.speed).padEnd(16)}
          </Text>
          <Text>ETA </Text>
          <Text color="yellowBright" bold>
            {formatEta(stats.etaSeconds)}
          </Text>
        </Box>
        <Box>
          <Text>Downloaded </Text>
          <Text color="whiteBright">
            {formatBytes(stats.validatedBytes).padEnd(12)}
          </Text>
          <Text>Uploaded </Text>
          <Text color="gray">{formatBytes(stats.bytesUploaded)}</Text>
        </Box>
        <Box>
          <Text dimColor>Buffer </Text>
          <Text color="gray">
            {formatBytes(stats.bytesInProgress).padEnd(16)}
          </Text>
          <Text dimColor>Swarm </Text>
          <Text color="gray">
            {stats.peersConnectedCount} peers{" "}
            <Text dimColor>
              ({stats.seedPeers} seed / {stats.peersChoked} choke)
            </Text>
          </Text>
        </Box>
      </Box>

      {/* ── PEERS SECTION ── */}
      {topPeers.length > 0 && (
        <Box flexDirection="column" marginBottom={1}>
          <Text dimColor bold>
            Active Swarm
          </Text>
          {topPeers.map((p) => (
            <Box key={p.addr} marginLeft={2}>
              <Text dimColor>↳ </Text>
              <Text color="cyan">{p.peerName.substring(0, 18).padEnd(20)}</Text>
              <Text color="gray">{formatMbps(p.down.rate).padStart(12)}</Text>
            </Box>
          ))}
        </Box>
      )}

      {/* ── FOOTER STATS ── */}
      <Box>
        <Text dimColor>mem </Text>
        <Text color="magenta">{formatBytes(memory.heapUsed)}</Text>
        <Text dimColor> rss </Text>
        <Text color="magenta">{formatBytes(memory.rss)}</Text>
      </Box>
    </Box>
  );
}

export function startStatsUI(stats: Stats) {
  render(<App stats={stats} />);
}
